# Desktop build of the app's AI reader, used in CI to run the real model on a synthetic invoice.
cmake_minimum_required(VERSION 3.22.1)
project(receipt_ai_cli C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_DIR "${CMAKE_CURRENT_LIST_DIR}/../../third_party/llama.cpp")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
foreach(opt LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_OPENSSL MTMD_VIDEO)
    set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_DIR}" llama-build)
add_executable(receipt-ai-cli main.cpp ../../app/src/main/cpp/receipt_ai.cpp)
target_include_directories(receipt-ai-cli PRIVATE ../../app/src/main/cpp "${LLAMA_DIR}/include" "${LLAMA_DIR}/ggml/include" "${LLAMA_DIR}/tools/mtmd" "${LLAMA_DIR}/vendor")
target_link_libraries(receipt-ai-cli PRIVATE llama mtmd ggml)
